# PROPOSED, NOT DECIDED. Operating system packages of the Alpine base layer that declare GPL or
# LGPL. The strict reading of T11 (only the Java runtime allowed) fails both images on these, so
# the scan cannot pass without a decision from the owner. Both workflows read this file, so
# removing it from their --allow lists makes the scan fail on them again.
#
# Why a list and not "apk:*": each line is a package the images ship today, matched by its
# declared license text, so a new package or a changed license shows up in the report.
# These are separate programs and shared libraries that the images contain and that glassops
# runs as commands (git, coreutils, busybox). They are not linked into glassops.
apk:acl-libs = LGPL-2.1-or-later AND GPL-2.0-or-later
apk:alpine-baselayout = GPL-2.0-only
apk:alpine-baselayout-data = GPL-2.0-only

apk:busybox = GPL-2.0-only

apk:coreutils-sha512sum = GPL-3.0-or-later

apk:git = GPL-2.0-only
apk:git-init-template = GPL-2.0-only
apk:java-common = GPL-2.0-or-later
apk:libapk = GPL-2.0-only
apk:libattr = LGPL-2.1-or-later
apk:libgcc = GPL-2.0-or-later AND LGPL-2.1-or-later
apk:libidn2 = GPL-2.0-or-later OR LGPL-3.0-or-later
